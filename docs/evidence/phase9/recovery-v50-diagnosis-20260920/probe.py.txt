from pathlib import Path
import sys,time,json,hashlib
root=Path.cwd();sys.path.insert(0,str(root));started=time.perf_counter()
import numpy as np
from workflows.identify_sparse_world_v30 import load_fit_cache
from koopman.bounded_mpc_v44 import load_fit_domains
from koopman.bounded_feedback_v47 import PreparedTrackingMap
out=root/'docs/evidence/phase9/recovery-v50-diagnosis-20260920'
source=root/'docs/evidence/phase9/recovery-v50-worker-20260920/result.json'
protocol=json.loads((out/'protocol.json').read_text())
assert hashlib.sha256(source.read_bytes()).hexdigest()==protocol['source_result_sha256']
result=json.loads(source.read_text());rows=[]
handoff=json.loads((root/'docs/evidence/phase8_4/server-projected-formal-v38-r23/prediction-control-handoff-v2.json').read_text())
domains=load_fit_domains(root,Path(handoff['frozen_source_directory'])/handoff['eligible_models']['nonlinear__pooled']['path'])
episodes=load_fit_cache(root)
for row in result['rows']:
    name=row['configuration']
    if name not in protocol['configurations']:continue
    answer=row['reply']['payload'];assert answer['reason']=='selected_command_constraints'
    baseline=np.array(answer['recovery']['commands'],dtype=float);chosen=baseline.copy()
    selected=answer['selected_index'];domain=domains[name];old=np.array(answer['metadata']['previous_command'])
    if selected:
        axis=(selected-1)//2;sign=(-1,1)[(selected-1)%2];chosen[8:,axis]+=sign*.002
        for t in range(8,20):
            prior=old if t==0 else chosen[t-1]
            chosen[t]=np.clip(chosen[t],domain.command_lower,domain.command_upper)
            chosen[t]=np.clip(chosen[t],prior-.01,prior+.01)
    chosen=chosen.astype(np.float32)
    c=next(e.context for e in episodes if e.case['configuration']==name)
    steady=PreparedTrackingMap(name,c);checks=[]
    for i,u in enumerate(chosen):
        record=steady.inspect(u,answer['recovery']['requested_wrench'],domain.command_lower,domain.command_upper)
        if not record['command_constraints_accepted']:
            checks.append(dict(index=i,command=u,inspection=record))
    rows.append(dict(configuration=name,selected_index=selected,failures=checks))
assert len(rows)==2 and time.perf_counter()-started<30
payload=dict(rows=rows,seconds=time.perf_counter()-started,new_predictions=0,new_simulation_intervals=0)
(out/'result.json').write_text(json.dumps(payload,default=lambda a:a.tolist(),indent=2),encoding='utf-8')
for row in rows:
    print(json.dumps(dict(configuration=row['configuration'],selected_index=row['selected_index'],
        failures=[dict(index=f['index'],distance=f['inspection']['minimum_deadzone_distance_pwm'],
            headroom=f['inspection']['pwm_headroom']) for f in row['failures']])))
