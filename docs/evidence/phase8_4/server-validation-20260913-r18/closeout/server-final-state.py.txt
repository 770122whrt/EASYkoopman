from pathlib import Path
import json,hashlib,subprocess,sys
root=Path('/root/EASYkoopman-phase8-4-validation-20260913-r18')
transfer=Path('/root/phase84-transfer-20260913/validation-r18');sys.path.insert(0,str(root))
from workflows.validation_release_v30 import verify_release
read=lambda p:json.loads(p.read_text())
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
source=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
verify_release(root,transfer,source)
dirty=subprocess.check_output(['git','status','--porcelain=v1','--untracked-files=all'],cwd=root,text=True).strip()
assert not dirty
budget=read(transfer/'budget.json');prior=read(transfer/'prior-budget.json')
assert budget['attempts'][:32]==prior['attempts']
assert abs(sum(a['charged_seconds'] for a in budget['attempts'])-budget['charged_seconds'])<1e-6
status=read(transfer/'validation/stage-status.json')
result={'status':status['status'],'source_commit':source,'source_files_verified':len(read(root/'SOURCE_MANIFEST.json')['files_sha256']),
    'native_attempts_total':len(budget['attempts']),'validation_native_attempts':len(budget['attempts'])-32,
    'accepted_validation':len(status['accepted']),'rejected_validation':status['rejected'],
    'all_native_exit_zero':all(a.get('status')=='exited' and a.get('native_exit')==0 for a in budget['attempts']),
    'collector_seconds_total':budget['charged_seconds'],'validation_collector_seconds':budget['charged_seconds']-prior['charged_seconds'],
    'transfer_bytes_total':sum(p.stat().st_size for p in transfer.parent.rglob('*') if p.is_file()),
    'gpu_compute_processes':subprocess.check_output(['nvidia-smi','--query-compute-apps=pid,process_name','--format=csv,noheader'],text=True).strip(),
    'frozen_models':34,'validation_refits':0,'test_access':False,'model_handoff':False,
    'validation_freeze_sha256':sha(transfer/'validation-freeze.json')}
(transfer/'server-final-state.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
