from pathlib import Path
import hashlib,json,tarfile
import numpy as np
from workflows.control_seam_v23 import ControlKernel
from workflows.pilot_control_v24 import commands
from workflows.validate_control_trace_v23 import validate_interval
from easyuuv_nc.control_v24 import ActuatorState
root=Path.cwd(); base=root/'docs/evidence/phase8_4/fresh-pilot-20260912'
archive=base/'failed-r6/pilot-r6-failure.tar.gz'
sha='66828c4c01e27bad3af939035b0d58f8f59ca8077af16a0991b0ef75e0c83057'
assert hashlib.sha256(archive.read_bytes()).hexdigest()==sha
raw=base/'failed-r6/raw';raw.mkdir(exist_ok=False)
with tarfile.open(archive) as tar:
 for m in tar.getmembers():
  assert (m.isfile() or m.isdir()) and (raw/m.name).resolve().is_relative_to(raw.resolve())
 tar.extractall(raw,filter='data')
inventory={p.relative_to(raw).as_posix():{'bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()} for p in sorted(raw.rglob('*')) if p.is_file()}
transfer=raw/'phase84-transfer-20260912/pilot-r6'
assert (transfer/'runner.exit_status').read_text().strip()=='1'
assert (transfer/'pilot-base-8411-prbs.exit_status').read_text().strip()=='0'
d=json.loads(next(raw.glob('EASYkoopman-*/tmp/phase8_4/pilot-base-8411-prbs/trace.json')).read_text())
assert d['status']=='failed_exploratory_v24' and len(d['substeps'])==242 and len(d['boundary_states'])==120
for i in range(121):validate_interval(d['substeps'][2*i:2*i+2],configuration='base')
clock=[r['actuator_update']['end_time_s'][0] for r in d['substeps'][-2:]]
assert abs(clock[1]-clock[0]-1/120)>1e-7
cases=[]
for request in sorted((root/'.pytest-tmp/phase84-pilot-requests').glob('*.json')):
 q=json.loads(request.read_text());k=ControlKernel(q['configuration'])
 estimate=ActuatorState(k.env._num_thrusters,tau=k.tau,dt=k.dt,clock='float32_accumulated_v1')
 speed_error=wrench_error=0.
 for u in commands(q):
  for _ in range(2):
   pwm=k.command(u,pre_tam=True)['pwm'];pred=estimate.advance_pwm(pwm);actual=k.advance(pwm)
   wrench=k.B.numpy()@(k.env.cfg.rotor_constant*np.abs(pred)*pred)
   speed_error=max(speed_error,float(np.max(np.abs(pred-actual['speed']))))
   wrench_error=max(wrench_error,float(np.max(np.abs(wrench-actual['wrench']))))
 assert speed_error<1e-3 and wrench_error<1e-3
 cases.append({'run_id':q['run_id'],'speed_max_abs':speed_error,'wrench_max_abs':wrench_error})
assert len(cases)==18
report={'status':'clock_bug_reproduced_and_focused_fix_checked','archive_sha256':sha,'failed_trace_status':d['status'],'native_case_exit':0,'semantic_batch_exit':1,'complete_intervals':120,'attempted_intervals':121,'model_fit_rows_used':0,'last_clock_values':clock,'last_tick_absolute_deviation':abs(clock[1]-clock[0]-1/120),'new_validator_all_recorded_intervals_pass':True,'relabel_failed_trace':False,'inventory_files':len(inventory),'cpu_only_fixed_request_actuator_checks':cases,'max_speed_error':max(c['speed_max_abs'] for c in cases),'max_wrench_error':max(c['wrench_max_abs'] for c in cases),'limits':'CPU actuator/control only, not a new Isaac hull run'}
(base/'failed-r6/inventory.json').write_text(json.dumps(inventory,indent=2)+'\n')
(base/'clock-diagnosis.json').write_text(json.dumps(report,indent=2)+'\n')
(base/'clock-audit-script.py.txt').write_bytes(Path(__file__).read_bytes())
print(json.dumps(report,indent=2))
