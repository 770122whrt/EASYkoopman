"""Post-validation model-form audit, using fixed fit states only; no refitting."""
from pathlib import Path
import sys,json,hashlib,time
sys.path.insert(0,str(Path.cwd()))
import numpy as np
from koopman.projected_edmd_v24 import PhysicalContext
from koopman.physical_terms_v26 import state_terms
from koopman.physical_prediction_v29 import advance_old_frame_velocity
from workflows.identify_sparse_world_v30 import from_record

root=Path.cwd();read=lambda p:json.loads(p.read_text(encoding='utf8'));sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
base=root/'source/results/phase8.4-sparse-world-pilot-v30-20260913';fit=root/'docs/evidence/phase8_4/identification-fit-20260913-r17'
freeze=read(root/'docs/evidence/phase8_4/validation-package-20260913-r19/validation-freeze.json')
inventory=read(fit/'cache-inventory.json')['files_sha256'];ledger=fit/'analysis-budget.json';budget=read(ledger)
reserve=60.;assert budget['charged_seconds']+reserve<=budget['limit_seconds']
attempt={'stage':'post_validation_v30_parameter_form_equivalence','status':'reserved','charged_seconds':reserve}
budget['attempts'].append(attempt);budget['charged_seconds']+=reserve;ledger.write_text(json.dumps(budget,indent=2)+'\n',encoding='utf8')
started=time.monotonic()
try:
    rows=[]
    for model_id,digest in freeze['models'].items():
        path=base/'models'/(model_id+'.json');assert sha(path)==digest;record=read(path);model=from_record(record)
        errors=[];comparisons=0
        for metadata in sorted(fit.glob('i29-*-fit-*.json')):
            assert sha(metadata)==inventory[metadata.name];e=read(metadata);cfg=e['case']['configuration']
            if record['scope'].startswith('local-') and model_id!='%s__local-%s'%(record['family'],cfg):continue
            if record['scope'].startswith('heldout-') and model_id!='%s__heldout-%s'%(record['family'],cfg):continue
            arrays=metadata.with_suffix('.npz');assert sha(arrays)==inventory[arrays.name]
            assert e['case']['role']=='fit';context=PhysicalContext(**e['context'])
            with np.load(arrays,allow_pickle=False) as data:
                ids=[0,256,384,512];x=data['states'][ids].copy();measured=data['acceleration'][ids].copy()
            terms=state_terms(x,context)
            for a in (measured,np.zeros_like(measured),-measured):
                # Direct physical parameter model, with no lift or A multiplication.
                rate=a+np.c_[terms['net_buoyancy'],terms['restoring']]+terms['linear_drag']
                rate+=terms['quadratic_drag']*model.quadratic-x[:,5:]*model.damping
                rate[:,3:]+=terms['gyro']
                nu=x[:,5:]+rate/120;nu[:,3:]*=1-model.angular_damping/120
                direct=advance_old_frame_velocity(x,nu)
                lifted=model(x,a,context)
                errors.append(float(np.max(np.abs(direct-lifted))));comparisons+=len(x)
        assert comparisons>0 and max(errors)<1e-12,(model_id,max(errors))
        rows.append({'model_id':model_id,'state_input_comparisons':comparisons,'maximum_state_difference':max(errors)})
    result={'status':'projected_v30_matches_direct_physical_parameter_model_on_fixed_fit_probes','models':len(rows),
        'state_input_comparisons':sum(r['state_input_comparisons'] for r in rows),'maximum_state_difference':max(r['maximum_state_difference'] for r in rows),
        'fixed_physics_indices':[0,256,384,512],'input_variants':['recorded_acceleration','zero','negated_recorded_acceleration'],
        'records':rows,'new_fits':0,'new_simulation':0,'uses_validation_states':False,'model_handoff':False,
        'interpretation':'The operational six velocity rows plus the same pose integrator are an algebraic lifted representation of bounded physical-parameter dynamics. This probe does not establish an independent benefit from fitting the remaining lift rows.',
        'limitation':'Finite source-only probes corroborate the code-level algebra; shared physical terms and pose integration are not independent physics validation.',
        'script_sha256':sha(Path(__file__)),'model_sha256':freeze['models'],'fit_cache_inventory_sha256':sha(fit/'cache-inventory.json')}
    target=root/'docs/evidence/phase8_4/validation-analysis-closeout-20260913-r19/parameter-equivalence.json'
    with target.open('x',encoding='utf8') as f:json.dump(result,f,indent=2);f.write('\n')
    with target.with_suffix('.py.txt').open('x',encoding='utf8') as f:f.write(Path(__file__).read_text(encoding='utf8'))
    attempt['status']='completed';print(json.dumps({k:result[k] for k in ('status','models','state_input_comparisons','maximum_state_difference')}))
finally:
    elapsed=time.monotonic()-started;budget['charged_seconds']+=elapsed-reserve;attempt['charged_seconds']=elapsed
    ledger.write_text(json.dumps(budget,indent=2)+'\n',encoding='utf8')
