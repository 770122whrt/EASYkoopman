from pathlib import Path
import hashlib,json,subprocess
root=Path('/root/EASYkoopman-phase8-4-identification-20260913-r17')
transfer=Path('/root/phase84-transfer-20260913/identification-r17')
read=lambda p:json.loads(p.read_text())
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
manifest=read(root/'SOURCE_MANIFEST.json')
assert not subprocess.check_output(['git','status','--porcelain=v1','--untracked-files=all'],cwd=root,text=True).strip()
source=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
assert source==(transfer/'expected-source-commit.txt').read_text().strip()
for name,h in manifest['files_sha256'].items():assert sha(root/name)==h,name
budget=read(transfer/'budget.json');assert len(budget['attempts'])==32
assert all(q['status']=='exited' and q['native_exit']==0 for q in budget['attempts'])
for stage,n in [('preflight',8),('fit',24)]:
    status=read(transfer/stage/'stage-status.json')
    assert len(status['accepted'])==n and not status['rejected'] and status['status']=='identification_stage_completed_pending_pullback'
result={'status':'preflight_fit_complete_source_unchanged','source_commit':source,'source_files_verified':len(manifest['files_sha256']),
    'native_attempts':32,'collector_seconds':budget['charged_seconds'],
    'transfer_bytes':sum(p.stat().st_size for p in transfer.rglob('*') if p.is_file()),
    'gpu_compute_processes':subprocess.check_output(['nvidia-smi','--query-compute-apps=pid,process_name','--format=csv,noheader'],text=True).strip(),
    'validation_attempted':(transfer/'validation').exists(),'model_fits':0}
assert not result['validation_attempted']
(transfer/'server-final-state.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
