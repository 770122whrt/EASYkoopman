"""Read-only linear-observable identity probe; no fit or promotion."""
from pathlib import Path
import os,sys,json,hashlib,time
for key in ('OMP_NUM_THREADS','OPENBLAS_NUM_THREADS','MKL_NUM_THREADS'):os.environ[key]='1'
sys.path.insert(0,str(Path.cwd()))
import numpy as np
from koopman.sparse_world_edmd_v30 import lift,feature_names
from workflows.identify_sparse_world_v30 import from_record,load_fit_cache
from workflows.controlled_lift_v33 import full_rollout
root=Path.cwd();read=lambda p:json.loads(p.read_text(encoding='utf8'));sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
base=root/'source/results/phase8.4-controlled-lift-v33-20260913';prior=root/'source/results/phase8.4-sparse-world-pilot-v30-20260913'
started=time.monotonic();name='nonlinear__pooled';parent=prior/'models'/(name+'.json');record=read(parent);ref=from_record(record)
fresh=base/'models'/(name+'.json');fitted=read(fresh);assert sha(parent)==fitted['parent_sha256']
names=feature_names('nonlinear');index={n:i for i,n in enumerate(names)};C=np.zeros((53,3));P=np.eye(53)
for k in range(3):
    C[index[f'vw{k}'],k]=1
    for j in range(3):C[index[f'axis_velocity_{j}_{k}'],k]=-1
for k in range(3):
    for j in range(3):P[:,index[f'axis_velocity_{j}_{k}']]+=C[:,k]/3
assert np.max(np.abs(P@C))<1e-15 and np.max(np.abs(P@P-P))<1e-15
np.testing.assert_array_equal(P[:,:16],np.eye(53)[:,:16])
episodes=load_fit_cache(root);features=np.concatenate([lift(e.states[[0,256,384,512]],e.context,'nonlinear') for e in episodes])
initial=float(np.max(np.abs(features@C)));rows=[]
for label,A in [('v30',ref.matrix),('v33',np.asarray(fitted['matrix']))]:
    eigenvalues,modes=np.linalg.eig(A.T);order=np.argsort(-np.abs(eigenvalues))[:3]
    eigen=[{'magnitude':float(abs(eigenvalues[i])),'linear_identity_violation':float(np.linalg.norm(modes[:,i]@C)/np.linalg.norm(modes[:,i]))} for i in order]
    e=episodes[0];mode='old' if label=='v30' else 'corrected'
    raw=full_rollout(e.states,e.acceleration,ref,e.context,A,mode,60,start_control=128)
    projected=full_rollout(e.states,e.acceleration,ref,e.context,A@P,mode,60,start_control=128)
    rows.append({'version':label,'initial_identity_max':initial,'after_one_operator_identity_max_on96fit_probes':float(np.max(np.abs(features@A@C))),
        'largest_row_eigenmodes':eigen,'spectral_radius_after_linear_identity_projection':float(max(abs(np.linalg.eigvals(A@P)))),
        'single_episode':e.case['run_id'],'raw':raw,'linear_identity_projected':projected})
result={'status':'source_only_exact_linear_observable_identity_audit','identity':'vw[k]=sum_j(axis_velocity[j,k])',
    'records':rows,'physics_readout_unchanged_by_linear_projection':True,'new_fits':0,'new_simulations':0,'validation_access':False,
    'scope':'96 fixed fit-state one-step probes; one existing base-fit episode, all legal60-control origins; posthoc mechanism intervention only.',
    'model_handoff':False,'input_arrays_changed':False,'frozen_model_bytes_changed':False,
