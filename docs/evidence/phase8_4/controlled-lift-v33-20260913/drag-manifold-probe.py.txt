"""One fixed fit origin: can an unconstrained lifted drag observable add energy?"""
from pathlib import Path
import os,sys,json,hashlib,time
for key in ('OMP_NUM_THREADS','OPENBLAS_NUM_THREADS','MKL_NUM_THREADS'):os.environ[key]='1'
sys.path.insert(0,str(Path.cwd()))
import numpy as np
from koopman.sparse_world_edmd_v30 import lift,feature_names
from koopman.controlled_lift_v33 import latent_step
from workflows.identify_sparse_world_v30 import load_fit_cache,from_record
from workflows.identification_prediction_v32 import valid_predictions
root=Path.cwd();read=lambda p:json.loads(p.read_text(encoding='utf8'));sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
started=time.monotonic();prior=root/'source/results/phase8.4-sparse-world-pilot-v30-20260913/models/nonlinear__pooled.json'
fresh=root/'source/results/phase8.4-controlled-lift-v33-20260913/models/nonlinear__pooled.json'
parent=read(prior);candidate=read(fresh);assert sha(prior)==candidate['parent_sha256'];ref=from_record(parent)
e=load_fit_cache(root)[0];names=feature_names('nonlinear');ids=[i for i,n in enumerate(names) if 'quadratic' in n]
index={n:i for i,n in enumerate(names)};linear_ids=[[index[f'axis_quadratic_{j}_{k}'] for k in range(3)] for j in range(3)]
angular_ids=[index[f'angular_quadratic_{j}'] for j in range(3)];records=[]
def power(z):
    drag=sum(ref.quadratic[j]*z[:,linear_ids[j]] for j in range(3))
    rotational=ref.quadratic[3:]*z[:,angular_ids]
    return float(e.context.mass*np.sum(z[:,10:13]*drag)+np.sum(e.context.inertia*z[:,13:16]*rotational))
for version,A,mode in [('v30',ref.matrix,'old'),('v33',np.asarray(candidate['matrix']),'corrected')]:
    for intervention in ('none','relift_quadratic_observables_only'):
        x=e.states[256:257].copy();z=lift(x,e.context,'nonlinear');trace=[];failure=None
        for tick in range(120):
            physical=lift(x,e.context,'nonlinear')
            if intervention!='none':z[:,ids]=physical[:,ids]
            trace.append({'physics_tick_from_origin':tick,'quadratic_drag_power_from_latent_W':power(z),
                'quadratic_drag_power_from_state_W':power(physical),'state':x[0].tolist()})
            with np.errstate(over='ignore',invalid='ignore',divide='ignore'):
                next_z,y=latent_step(z,x,e.acceleration[256+tick:257+tick],e.context,ref,A,mode)
            if not valid_predictions(y).all():failure=tick+1;break
            z=next_z;x=y
        positive=next((r for r in trace if r['quadratic_drag_power_from_latent_W']>1e-10),None)
        records.append({'version':version,'intervention':intervention,'completed_physics_ticks':len(trace) if failure is None else failure-1,
            'first_failure_physics_tick':failure,'first_positive_drag_power':positive,'trace':trace})
result={'status':'source_only_fixed_origin_drag_sign_intervention','run_id':e.case['run_id'],'origin_control':128,'horizon_control':60,
    'records':records,'new_fits':0,'new_simulations':0,'validation_access':False,'model_handoff':False,
    'scope':'One predetermined base-fit origin with fixed recorded accelerations; quadratic-only projection is a posthoc mechanism intervention, not a selected candidate or general performance claim.',
    'script_sha256':sha(Path(__file__)),'model_sha256':{'v30':sha(prior),'v33':sha(fresh)},'seconds':time.monotonic()-started}
out=root/'docs/evidence/phase8_4/controlled-lift-v33-20260913'
with (out/'drag-manifold-probe.json').open('x',encoding='utf8') as f:json.dump(result,f,indent=2);f.write('\n')
with (out/'drag-manifold-probe.py.txt').open('x',encoding='utf8') as f:f.write(Path(__file__).read_text(encoding='utf8'))
print(json.dumps({'seconds':result['seconds'],'rows':[{k:r[k] for k in ('version','intervention','completed_physics_ticks','first_failure_physics_tick')}|{'first_positive_drag_tick':None if r['first_positive_drag_power'] is None else r['first_positive_drag_power']['physics_tick_from_origin']} for r in records]}))
