from pathlib import Path
import json,subprocess,time,sys
root=Path.cwd();mode=sys.argv[1]
pkg=root/'docs/evidence/phase8_4/feedback-package-20260913-r14'
tests=json.loads((root/'docs/evidence/phase8_4/calibration-package-20260913-r11/root-tests.json').read_text())['tests']+['tests/test_feedback_v28.py','tests/test_actuator_replay_v28.py','tests/test_collector_supervisor_v28.py']
work=root if mode=='root' else Path(json.loads((pkg/'package.json').read_text())['clone_path'])
args=[str(root/'.venv/Scripts/python.exe'),'-X','utf8','-B','-m','pytest',*tests,'-q','-p','no:cacheprovider','--basetemp',str(root/('.pytest-tmp/feedback-r14-tests-'+mode))]
t=time.monotonic();r=subprocess.run(args,cwd=work,capture_output=True)
out=(r.stdout+r.stderr).decode('utf8',errors='replace')
(pkg/(mode+'-tests.txt')).write_text(out,encoding='utf8')
result={'scope':mode,'exit_code':r.returncode,'seconds':time.monotonic()-t,'tests':tests,'summary':out.strip().splitlines()[-1]}
(pkg/(mode+'-tests.json')).write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result));sys.exit(r.returncode)
