"""Logged-wrench diagnostic; measured next attitude is not a predictor input."""
import json,hashlib
from pathlib import Path
import numpy as np
from koopman.projected_edmd_v24 import rotation
root=Path.cwd();source=root/'docs/evidence/phase8_4/server-fresh-pilot-20260912-r7/raw/results'
out=root/'docs/evidence/phase8_4/known-input-semantics-20260912';out.mkdir(exist_ok=False)
records=[]
for p in sorted(source.glob('*/trace.json')):
 d=json.loads(p.read_text());errors=[]
 for row in d['substeps']:
  x=np.array(row['before']['state_11']);y=np.array(row['state_after_physics_11'])
  R=rotation(x[:,1:5])[0];S=rotation(y[:,1:5])[0];t=row['command'];b=row['before']['backend']
  mass=float(np.array(b['mass_kg']).item());I=np.array(b['inertia_9']).reshape(3,3).diagonal()
  F=np.array(t['_thrust']).reshape(3);T=np.array(t['_moment']).reshape(3);w=x[0,8:];v=x[0,5:8]
  world_v=R@v+(R@F/mass+[0,0,-9.81])/120
  diagnosed=S.T@world_v
  explicit_v=v+(F/mass+R.T@np.array([0,0,-9.81])-np.cross(w,v))/120
  explicit_w=w+(T/I-np.cross(w,I*w)/I)/120
  errors.append(np.r_[diagnosed-y[0,5:8],explicit_v-y[0,5:8],explicit_w-y[0,8:]])
 a=np.asarray(errors)
 records.append({'run_id':d['request']['run_id'],'configuration':d['request']['configuration'],'trace_sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'rows':len(a),'max_abs':np.abs(a).max(0).tolist(),'rmse':np.sqrt(np.mean(a*a,axis=0)).tolist()})
report={'status':'logged_wrench_coordinate_and_discretization_diagnostic','fits':0,'new_simulation_intervals':0,'physics_rows':sum(r['rows'] for r in records),'error_channels':['translation_using_measured_next_R_x','translation_using_measured_next_R_y','translation_using_measured_next_R_z','explicit_body_Euler_vx','explicit_body_Euler_vy','explicit_body_Euler_vz','explicit_body_Euler_wx','explicit_body_Euler_wy','explicit_body_Euler_wz'],'max_abs_all_rows':np.max([r['max_abs'] for r in records],axis=0).tolist(),'cases':records,'gyroscopic_source_enabled':True,'asset_source_sha256':hashlib.sha256((root/'easyuuv_nc/env/assets/warpauv.py').read_bytes()).hexdigest(),'limits':['measured next attitude and totalwrench are diagnostictruth, unavailable in recursive predictor','explicitbodyEuler is approximation, not exact PhysXrotationalintegration','no new model forecast/learning score','known instantaneous input map does not mean entire discretevelocity map is dtI'],'reference':'https://nvidia-omniverse.github.io/PhysX/physx/5.4.0/docs/RigidBodyDynamics.html#gyroscopic-forces'}
(out/'diagnosis.json').write_text(json.dumps(report,indent=2)+'\n');(out/'analysis-script.py.txt').write_bytes(Path(__file__).read_bytes())
print(json.dumps({k:v for k,v in report.items() if k!='cases'},indent=2))
