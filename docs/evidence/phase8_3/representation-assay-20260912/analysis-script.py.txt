def metrics(predicted,actual):
 return {'depth':float(np.sqrt(np.mean((predicted[...,0]-actual[...,0])**2))),
         'attitude':float(np.sqrt(np.mean(so3_geodesic_radians_v21(predicted[...,1:5],actual[...,1:5])**2))),
         'linear_velocity':float(np.sqrt(np.mean((predicted[...,5:8]-actual[...,5:8])**2))),
         'angular_velocity':float(np.sqrt(np.mean((predicted[...,8:11]-actual[...,8:11])**2)))}

def evaluate(model,ep,variant,horizon):
 endpoints=[];paths=[];failures=[]
 memory=ep['speed' if variant.endswith('speed') else 'proxy'] if variant!='persistence' else None
 for start in range(33-horizon):
  if model is None:
   predictions=np.tile(ep['states'][start],(horizon,1))
  else:
   r=rollout(model,ep['states'][start],memory[start:start+horizon],ep['controls'][start:start+horizon],variant=variant,mask=ep['meta']['mask'])
   if r['status']!='success':failures.append({'start':start,**{k:v for k,v in r.items() if k!='predictions'}});continue
   predictions=r['predictions']
  endpoints.append(predictions[-1]);paths.append(predictions)
 row={'horizon':horizon,'origins':33-horizon,'successes':len(endpoints),'failures':failures,'endpoint_rmse':None,'path_rmse':None}
 if not failures:
  row['endpoint_rmse']=metrics(np.asarray(endpoints),ep['states'][horizon:])
  truth=np.asarray([ep['states'][s+1:s+1+horizon] for s in range(33-horizon)])
  row['path_rmse']=metrics(np.asarray(paths),truth)
 return row

def correlation(residual,history):
 r=residual-residual.mean(0);h=history-history.mean(0)
 den=np.linalg.norm(r,axis=0)[:,None]*np.linalg.norm(h,axis=0)[None,:]
 result=np.divide(r.T@h,den,out=np.zeros_like(den),where=den>1e-12)
 return {'matrix':result.tolist(),'undefined_zero_variance':(den<=1e-12).tolist(),'max_abs':float(np.max(np.abs(result)))}
